      this.company;

          </p>

      </>

PART 4 — Active Link Highlighting in Navbar
Step 1: Update Navbar.tsx

Replace your current Navbar.tsx code with the following:

📄 src/components/Navbar.tsx
import { NavLink } from "react-router-dom";

function Navbar() {
  const linkClass = ({ isActive }: { isActive: boolean }) =>
    isActive
      ? "text-yellow-300 font-semibold border-b-2 border-yellow-300 pb-1"
      : "hover:text-yellow-300 transition-colors";

  return (
    <nav className="bg-indigo-600 text-white p-4 flex justify-center gap-8 shadow-md">
      <NavLink to="/" className={linkClass}>Home</NavLink>
      <NavLink to="/about" className={linkClass}>About</NavLink>
      <NavLink to="/contact" className={linkClass}>Contact</NavLink>
    </nav>
  );
}

export default Navbar;

🧠 Explanation

NavLink is like Link, but it automatically gives you access to an isActive boolean.

We use a function (linkClass) to conditionally style the active link.

The active route link will have:

A yellow underline (border-b-2 border-yellow-300)

Bold text (font-semibold)

Highlighted color (text-yellow-300)

Step 2: Run and Test
npm run dev


Now navigate between Home, About, and Contact — the active page’s link will be highlighted beautifully ✨

💡 Optional Enhancements

You can make it even more interactive:

hover:bg-indigo-700 px-3 py-1 rounded-md


Example (if you want a button-like navbar):

const linkClass = ({ isActive }: { isActive: boolean }) =>
  isActive
    ? "bg-yellow-300 text-indigo-700 px-3 py-1 rounded-md font-bold"
    : "hover:bg-indigo-700 px-3 py-1 rounded-md transition-all";

PART 5 — Create a 404 Page (Not Found)
Step 1: Create a new file

Create a new file in your src/pages folder:

import { Link } from "react-router-dom";

function NotFound() {
  return (
    <div className="flex flex-col items-center justify-center min-h-screen bg-gray-100 text-center">
      <h1 className="text-6xl font-bold text-red-500 mb-4">404</h1>
      <h2 className="text-2xl font-semibold mb-2">Oops! Page not found 😢</h2>
      <p className="text-gray-600 mb-6">
        The page you’re looking for doesn’t exist or has been moved.
      </p>
      <Link
        to="/"
        className="bg-indigo-600 text-white px-5 py-2 rounded-md hover:bg-indigo-700 transition-colors"
      >
        Go Back Home
      </Link>
    </div>
  );
}

export default NotFound;

Step 2: Add Route in App.tsx

Open src/App.tsx and update your routes like this:

import { Routes, Route } from "react-router-dom";
import Navbar from "./components/Navbar";
import Home from "./pages/Home";
import About from "./pages/About";
import Contact from "./pages/Contact";
import NotFound from "./pages/NotFound"; // 👈 import new page

function App() {
  return (
    <div className="min-h-screen bg-gray-100">
      <Navbar />
      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/about" element={<About />} />
        <Route path="/contact" element={<Contact />} />
        <Route path="*" element={<NotFound />} /> {/* 👈 Catch-all route */}
      </Routes>
    </div>
  );
}

export default App;
